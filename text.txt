    <script>
        let cart = []; let navStack = ['main-menu']; const TIKTOK_URL = "https://www.tiktok.com/@codrl2";

        function previewImage(input) {
            const preview = document.getElementById('imagePreview');
            const placeholder = document.getElementById('uploadPlaceholder');
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function(e) { preview.src = e.target.result; preview.style.display = 'block'; placeholder.style.display = 'none'; };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function addToCart(name, price) { cart.push({ name, price }); updateCartUI(); showToast(`تمت إضافة ${name}`); }
        function addServiceToCart(n, p) { addToCart(n, p); closeModal('taskModal'); document.getElementById('imagePreview').style.display='none'; document.getElementById('uploadPlaceholder').style.display='block'; }
        function removeFromCart(i) { cart.splice(i, 1); updateCartUI(); }

        function updateCartUI() {
            const container = document.getElementById('cartItemsContainer'); const badge = document.getElementById('cartBadge'); const totalDisplay = document.getElementById('cartTotalPrice');
            badge.innerText = cart.length;
            if (cart.length === 0) { container.innerHTML = '<p style="text-align:center; color:#666;">سلتك فارغة</p>'; totalDisplay.innerText = "0.00 $"; return; }
            container.innerHTML = cart.map((item, i) => `<div class="cart-item"><span>${item.name}</span><div>${item.price.toFixed(2)} $ <span class="trash-btn" onclick="removeFromCart(${i})">🗑️</span></div></div>`).join('');
            totalDisplay.innerText = cart.reduce((sum, item) => sum + item.price, 0).toFixed(2) + " $";
        }

        function checkout() {
            if (cart.length === 0) return;
            let msg = "طلب جديد من متجر codrl:\n";
            cart.forEach((item, i) => msg += `${i+1}. ${item.name} ($${item.price.toFixed(2)})\n`);
            msg += `\nإجمالي الحساب: $${document.getElementById('cartTotalPrice').innerText}`;
            navigator.clipboard.writeText(msg).then(() => { alert("✅ تم نسخ طلبك! الصقه في الخاص بتيك توك."); window.open(TIKTOK_URL, '_blank'); });
        }

        function showMenu(id, title) { hideAll(); document.getElementById(id).style.display = 'grid'; document.getElementById('backBtn').style.display = 'block'; document.getElementById('sectionTitle').innerText = title; navStack.push(id); }
        function goBack() { if (navStack.length > 1) { navStack.pop(); const prev = navStack[navStack.length - 1]; hideAll(); document.getElementById(prev).style.display = 'grid'; if (prev === 'main-menu') { document.getElementById('backBtn').style.display = 'none'; document.getElementById('sectionTitle').innerText = 'الأقسام المتاحة'; } } }
        function hideAll() { ['main-menu', 'fortnite-submenu', 'roblox-submenu', 'gta-submenu', 'sprites-gallery'].forEach(id => document.getElementById(id).style.display = 'none'); }
        function openModal(id) { document.getElementById(id).style.display = 'flex'; }
        function closeModal(id) { document.getElementById(id).style.display = 'none'; }
        function updateLvl(v) { document.getElementById('lvlRes').innerText = (v * 0.1).toFixed(2); }
        function addLvlToCart() { const v = document.getElementById('lvlInput').value; if(v > 0) { addToCart(`تلفيل ${v} لفل`, v * 0.1); closeModal('lvlModal'); } }
        function showToast(txt) { const t = document.getElementById('toast'); t.innerText = `✅ ${txt}`; t.classList.add('show'); setTimeout(() => t.classList.remove('show'), 3000); }
        window.onclick = function(e) { if(e.target.className === 'modal') { closeModal('cartModal'); closeModal('lvlModal'); closeModal('taskModal'); } }
    </script>
